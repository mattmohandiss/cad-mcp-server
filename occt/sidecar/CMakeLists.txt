cmake_minimum_required(VERSION 3.20)
project(occt_sidecar_prototype LANGUAGES C CXX)
if(DEFINED COSMO_ROOT)
  # Override cached compiler-detection values when updating an existing build.
  set(CMAKE_AR "${CMAKE_BINARY_DIR}/cosmoar-wrapper")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OCCT_SOURCE_DIR "" CACHE PATH "OCCT V8.0.1 source directory")
set(JSON_INCLUDE_DIR "" CACHE PATH "Directory containing json.hpp")
if(NOT EXISTS "${OCCT_SOURCE_DIR}/CMakeLists.txt" OR NOT EXISTS "${JSON_INCLUDE_DIR}/json.hpp")
  message(FATAL_ERROR "Set OCCT_SOURCE_DIR and JSON_INCLUDE_DIR")
endif()
set(BUILD_LIBRARY_TYPE Static CACHE STRING "" FORCE)
foreach(module FoundationClasses ModelingData ModelingAlgorithms Visualization ApplicationFramework DataExchange Draw)
  set(BUILD_MODULE_${module} OFF CACHE BOOL "" FORCE)
endforeach()
set(BUILD_ADDITIONAL_TOOLKITS "TKDESTEP;TKPrim;TKMesh" CACHE STRING "" FORCE)
set(BUILD_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_DOC_Overview OFF CACHE BOOL "" FORCE)
set(BUILD_DOC_RefMan OFF CACHE BOOL "" FORCE)
set(USE_FREETYPE OFF CACHE BOOL "" FORCE)
set(USE_XLIB OFF CACHE BOOL "" FORCE)
set(USE_TBB OFF CACHE BOOL "" FORCE)
set(USE_RAPIDJSON OFF CACHE BOOL "" FORCE)
set(USE_DRACO OFF CACHE BOOL "" FORCE)
set(USE_MMGR_TYPE NATIVE CACHE STRING "" FORCE)
set(BUILD_USE_PCH OFF CACHE BOOL "" FORCE)
add_subdirectory("${OCCT_SOURCE_DIR}" occt)
# OCCT's version-header helper uses CURRENT_BINARY_DIR while its include collector
# uses BINARY_DIR. Bridge those locations when embedding OCCT as a subproject.
file(COPY "${CMAKE_CURRENT_BINARY_DIR}/occt/include/opencascade/Standard_Version.hxx"
  DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/include/opencascade")
add_executable(occt-sidecar cpp/sidecar.cpp)
target_include_directories(occt-sidecar PRIVATE "${JSON_INCLUDE_DIR}" "${CMAKE_BINARY_DIR}/include/opencascade")
target_link_libraries(occt-sidecar PRIVATE TKDESTEP TKPrim TKMesh)
add_executable(generate-repeated-assembly-fixture EXCLUDE_FROM_ALL tests/generate_repeated_assembly.cpp)
target_include_directories(generate-repeated-assembly-fixture PRIVATE "${CMAKE_BINARY_DIR}/include/opencascade")
target_link_libraries(generate-repeated-assembly-fixture PRIVATE TKDESTEP TKPrim TKMesh)
if(SIDECAR_FULL_STATIC)
  target_link_options(occt-sidecar PRIVATE -static)
endif()
if(UNIX AND NOT APPLE AND NOT DEFINED COSMO_ROOT)
  add_executable(occt-sidecar-static cpp/sidecar.cpp)
  target_include_directories(occt-sidecar-static PRIVATE "${JSON_INCLUDE_DIR}" "${CMAKE_BINARY_DIR}/include/opencascade")
  target_link_libraries(occt-sidecar-static PRIVATE TKDESTEP TKPrim TKMesh)
  target_link_options(occt-sidecar-static PRIVATE -static)
endif()
